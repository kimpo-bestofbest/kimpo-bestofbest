window.settle = async function() { if (state.settled) { return msg("adminMessage", "이미 정산이 끝났습니다."); } if (!state.bets.length) { return msg("adminMessage", "참여 기록이 없습니다."); } const winner = Number(document.getElementById("winnerSelect").value); const winningBets = state.bets.filter(b => b.team === winner); if (!winningBets.length) { return msg("adminMessage", "우승 팀에 배팅한 사람이 없어 정산할 수 없습니다."); } // 전체 배팅 포인트 const totalPool = state.bets.reduce( (sum, b) => sum + Number(b.amount || 0), 0 ); // 우승 팀에 배팅된 포인트 const winningPool = winningBets.reduce( (sum, b) => sum + Number(b.amount || 0), 0 ); if (winningPool <= 0) { return msg("adminMessage", "배팅 금액이 올바르지 않습니다."); } // 배팅 비율에 따른 정산 금액 계산 const shares = winningBets.map((b, index) => { const exact = totalPool * Number(b.amount) / winningPool; return { bet: b, index, payout: Math.floor(exact), remainder: exact - Math.floor(exact) }; }); // 소수점으로 남은 포인트 분배 let remaining = totalPool - shares.reduce( (sum, s) => sum + s.payout, 0 ); const remainderOrder = [...shares].sort( (a, b) => b.remainder - a.remainder || a.index - b.index ); for (let i = 0; i < remaining; i++) { remainderOrder[i].payout += 1; } // 우승 팀 배팅자에게 포인트 지급 shares.forEach(s => { const user = state.users[s.bet.userId]; if (user) { user.points = Number(user.points || 0) + s.payout; } }); state.settled = true; state.winner = winner; render(); await saveState(); const winnerName = state.teams.find(t => t.id === winner)?.name || "우승 팀"; msg( "adminMessage", `${winnerName} 정산 완료! 전체 배팅 포인트 ${totalPool.toLocaleString()}P를 우승 팀 배팅 비율에 따라 분배했습니다.` ); };
